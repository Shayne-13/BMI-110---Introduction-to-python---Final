 214    def Login(self, strRole, strIdentification, strPassword):
 215        """
 216        Authenticate a user and load their permissions into the session.
 217
 218        Patients are not password-authenticated — any patient ID is accepted.
 219        On success, updates ``currentUserCredentials`` and ``currentUserPermissions``
 220        with the authenticated user's data.
 221
 222        Args:
 223            strRole (str): Role to authenticate as. Must be "Doctor", "Nurse", or "Patient".
 224            strIdentification (str): The user's ID / username.
 225            strPassword (str): The user's password (ignored for Patient logins).
 226
 227        Returns:
 228            Doctor | Nurse | Patient | bool: The authenticated role object on success,
 229                                             or False if authentication fails.
 230        """
 231        if strRole == "Doctor":
 232            found = self.doctors.get(strIdentification)
 233            if found and found._Doctor__password == strPassword:
 234                self.currentUserCredentials["Role"] = "Doctor"
 235                self.currentUserCredentials["Identification"] = strIdentification
 236                self.currentUserPermissions = found.permissions
 237                return found
 238        elif strRole == "Nurse":
 239            found = self.nurses.get(strIdentification)
 240            if found and found._Nurse__password == strPassword:
 241                self.currentUserCredentials["Role"] = "Nurse"
 242                self.currentUserCredentials["Identification"] = strIdentification
 243                self.currentUserPermissions = found.permissions
 244                return found
 245        elif strRole == "Patient":
 246            found = self.patients.get(strIdentification)
 247            if found:
 248                self.currentUserCredentials["Role"] = "Patient"
 249                self.currentUserCredentials["Identification"] = strIdentification
 250                self.currentUserPermissions = found.permissions
 251                return found
 252        return False
